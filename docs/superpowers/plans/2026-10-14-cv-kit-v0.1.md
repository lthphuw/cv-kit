# cv-kit v0.1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the v0.1 foundation of the `cv-kit` Claude Code plugin: manifests, a validator, a trailer-blocking commit hook, the rules installer, six skills, two agents and shared references.

**Architecture:** A single-repo marketplace plus plugin (`source: "./"`). Deterministic pieces (validator, hook, rules installer, research-claims linter) are small stdlib-plus-PyYAML Python scripts built test-first. Skills are short Markdown files that point at `shared/` references. Rules are not a plugin component, so they ship as source files that `install-rules` copies into a project, with a PreToolUse hook as the hard guard for commits.

**Tech Stack:** Python 3.10+, PyYAML, pytest, Claude Code plugin format, GitHub Actions.

**Spec:** `docs/superpowers/specs/2026-10-14-cv-kit-design.md`. Research behind it: `docs/research/01-models-and-skills.md`, `02-deep-research.md`, `03-packaging.md`.

## Global Constraints

- Claude Code only: no per-harness manifests, no `CLAUDE.md` at the plugin root.
- `version` lives only in `.claude-plugin/plugin.json` (`0.1.0`); the marketplace entry carries none.
- Skill rules: `name` equals its directory and matches `^[a-z0-9]+(-[a-z0-9]+)*$`; `description` is at most 1,024 characters and starts with `Use when`; `SKILL.md` is at most 500 lines (target 120); critical instructions come first.
- Skill names carry `cv-` where the bare name is ambiguous, never `cv-kit-`.
- A catalog row is `verified` or `UNVERIFIED`. A model id, arXiv id or licence that was not checked is never written as fact.
- Commit messages: subject plus optional body, no trailers of any kind (no `Co-Authored-By`, `Signed-off-by`, `Generated-by`, "Generated with …"), and no mention of who or what reviewed the change.
- Never run `git commit` until the user has seen a preview (branch, `git diff --cached --stat`, full message) and approved it. Approval for one commit does not cover the next. Never use `--no-verify`. Never push without the user's confirmation.
- Python scripts use the standard library plus `pyyaml`; no network access in tests.
- No absolute local paths (`/Users/…`, `C:\…`) in shipped files.

## Review Focus

1. A commit message supplied through `-F <file>` or a heredoc: the hook must scan the file's content and the heredoc text, not only `-m`.
2. `git commit --amend --no-edit` reuses an old message the hook cannot see; this is a known gap that the README states.
3. A catalog row marked `verified` whose arXiv id is malformed or missing: the validator must fail it.
4. `install-rules` into a project that already has a same-named rule file: the user's file survives unless `--force` is given.
5. A malformed or missing SKILL.md frontmatter: the validator reports the path and continues instead of crashing.

```python
"""Check the cv-kit house rules that `claude plugin validate` does not enforce."""
from __future__ import annotations

import json
import re
import sys
from pathlib import Path

import yaml

NAME_RE = re.compile(r"^[a-z0-9]+(-[a-z0-9]+)*$")
ARXIV_RE = re.compile(r"^\d{4}\.\d{4,5}(v\d+)?$")
LOCAL_PATH_RE = re.compile(r"/Users/|[A-Za-z]:\\")
LINK_RE = re.compile(r"\]\((?!https?://|#|mailto:)([^)#\s]+)")
SKILL_DIR_RE = re.compile(r"\$\{CLAUDE_SKILL_DIR\}/([^\s`)\"']+)")
SKILL_KEYS = {
    "name", "description", "when_to_use", "disable-model-invocation", "user-invocable", "paths",
    "allowed-tools", "argument-hint", "model", "effort", "context", "agent", "license",
    "compatibility", "metadata",
}
AGENT_IGNORED = {"permissionMode", "hooks", "mcpServers", "initialPrompt"}
CATALOG_STATUS = {"verified", "UNVERIFIED"}
SCAN_PATHS = ["skills", "agents", "hooks", "rules", "shared", ".claude-plugin", "README.md"]
SCAN_SUFFIXES = {".md", ".json", ".py", ".sh", ".yml", ".yaml"}


def read_frontmatter(path: Path) -> tuple[dict, str]:
    text = path.read_text(encoding="utf-8")
    match = re.match(r"---\n(.*?)\n---\n", text, re.S)
    if not match:
        raise ValueError("no frontmatter at line 1")
    data = yaml.safe_load(match.group(1))
    if not isinstance(data, dict):
        raise ValueError("frontmatter is not a mapping")
    return data, text


def _inside(root: Path, target: Path) -> bool:
    return target.exists() and root.resolve() in [target.resolve(), *target.resolve().parents]


def _check_links(md: Path, root: Path, text: str) -> list[str]:
    errors = []
    refs = LINK_RE.findall(text) + SKILL_DIR_RE.findall(text)
    for rel in refs:
        if not _inside(root, md.parent / rel):
            errors.append(f"{md}: reference {rel!r} does not resolve inside the plugin")
    return errors


def check_skills(root: Path) -> list[str]:
    errors: list[str] = []
    seen: dict[str, Path] = {}
    total = 0
    for directory in sorted(p for p in (root / "skills").glob("*") if p.is_dir()):
        if not (directory / "SKILL.md").exists():
            errors.append(f"{directory}: missing SKILL.md")
    for md in sorted(root.glob("skills/*/SKILL.md")):
        try:
            fm, text = read_frontmatter(md)
        except Exception as exc:  # malformed YAML must not stop the run
            errors.append(f"{md}: frontmatter error: {exc}")
            continue
        name, desc = str(fm.get("name", "")), str(fm.get("description", ""))
        total += len(desc)
        if name != md.parent.name or not NAME_RE.match(name) or len(name) > 64:
            errors.append(f"{md}: name {name!r} must equal its directory and match [a-z0-9-]")
        if name in seen:
            errors.append(f"{md}: duplicate name, also in {seen[name]}")
        seen[name] = md
        if not desc or len(desc) > 1024:
            errors.append(f"{md}: description empty or over 1024 characters ({len(desc)})")
        elif not desc.startswith("Use when"):
            errors.append(f"{md}: description must start with 'Use when'")
        for key in sorted(set(fm) - SKILL_KEYS):
            errors.append(f"{md}: unknown frontmatter key {key!r}")
        if text.count("\n") + 1 > 500:
            errors.append(f"{md}: more than 500 lines")
        errors += _check_links(md, root, text)
    if total > 8000:
        errors.append(f"skills: descriptions total {total} characters, budget is 8000")
    return errors


def check_agents(root: Path) -> list[str]:
    errors: list[str] = []
    skill_names = {p.parent.name for p in root.glob("skills/*/SKILL.md")}
    for md in sorted(root.glob("agents/*.md")):
        try:
            fm, _ = read_frontmatter(md)
        except Exception as exc:
            errors.append(f"{md}: frontmatter error: {exc}")
            continue
        name = str(fm.get("name", ""))
        if not name or ":" in name:
            errors.append(f"{md}: agent name {name!r} is empty or contains ':'")
        if not fm.get("description"):
            errors.append(f"{md}: missing description")
        for key in sorted(AGENT_IGNORED & set(fm)):
            errors.append(f"{md}: {key} is ignored for plugin agents")
        for skill in fm.get("skills") or []:
            if skill not in skill_names:
                errors.append(f"{md}: preloads unknown skill {skill!r}")
    return errors


def _load_json(path: Path, errors: list[str]) -> dict | None:
    if not path.exists():
        errors.append(f"{path}: missing")
        return None
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except json.JSONDecodeError as exc:
        errors.append(f"{path}: invalid JSON: {exc}")
        return None


def check_manifests(root: Path) -> list[str]:
    errors: list[str] = []
    plugin = _load_json(root / ".claude-plugin/plugin.json", errors)
    market = _load_json(root / ".claude-plugin/marketplace.json", errors)
    if plugin and market:
        if "version" not in plugin:
            errors.append(".claude-plugin/plugin.json: version is required (single source)")
        entries = {e.get("name"): e for e in market.get("plugins", [])}
        if plugin.get("name") not in entries:
            errors.append("marketplace.json: no plugin entry matches plugin.json name")
        for name, entry in entries.items():
            if "version" in entry:
                errors.append(f"marketplace.json: entry {name!r} must not set version")
    hooks_file = root / "hooks/hooks.json"
    if hooks_file.exists():
        hooks = _load_json(hooks_file, errors)
        if hooks is not None:
            if "hooks" not in hooks:
                errors.append(f"{hooks_file}: needs a top-level 'hooks' key")
            else:
                errors += _check_hook_paths(root, hooks_file, hooks["hooks"])
    return errors


def _check_hook_paths(root: Path, hooks_file: Path, events: dict) -> list[str]:
    errors = []
    for groups in events.values():
        for group in groups:
            for handler in group.get("hooks", []):
                for arg in handler.get("args", []):
                    if "${CLAUDE_PLUGIN_ROOT}" in arg:
                        target = root / arg.replace("${CLAUDE_PLUGIN_ROOT}/", "")
                        if not target.exists():
                            errors.append(f"{hooks_file}: {arg} does not exist")
    return errors


def check_rules(root: Path) -> list[str]:
    return [
        f"{md}: more than 200 lines"
        for md in sorted(root.glob("rules/*.md"))
        if md.read_text(encoding="utf-8").count("\n") + 1 > 200
    ]


def check_catalog(root: Path) -> list[str]:
    path = root / "shared/catalog.md"
    if not path.exists():
        return []
    errors, in_table = [], False
    for number, line in enumerate(path.read_text(encoding="utf-8").splitlines(), 1):
        if line.startswith("| Model |"):
            in_table = True
            continue
        if not line.startswith("|"):
            in_table = False
            continue
        if not in_table or set(line) <= set("|-: "):
            continue
        cells = [c.strip() for c in line.strip().strip("|").split("|")]
        where = f"{path}:{number}"
        if len(cells) != 8:
            errors.append(f"{where}: expected 8 cells, found {len(cells)}")
            continue
        status = cells[7]
        if status not in CATALOG_STATUS:
            errors.append(f"{where}: Status {status!r} must be verified or UNVERIFIED")
        elif status == "verified" and cells[2] != "—" and not ARXIV_RE.match(cells[2]):
            errors.append(f"{where}: verified row has invalid arXiv id {cells[2]!r}")
    return errors


def check_local_paths(root: Path) -> list[str]:
    errors = []
    for name in SCAN_PATHS:
        base = root / name
        files = [base] if base.is_file() else sorted(base.rglob("*")) if base.is_dir() else []
        for file in files:
            if file.is_file() and file.suffix in SCAN_SUFFIXES and LOCAL_PATH_RE.search(file.read_text(encoding="utf-8")):
                errors.append(f"{file}: contains a local path")
    return errors


def run_checks(root: Path) -> list[str]:
    errors: list[str] = []
    for check in (check_skills, check_agents, check_manifests, check_rules, check_catalog, check_local_paths):
        errors += check(root)
    return errors


def main(argv: list[str] | None = None) -> int:
    args = sys.argv[1:] if argv is None else argv
    root = Path(args[0]) if args else Path(__file__).resolve().parents[1]
    errors = run_checks(root)
    for error in errors:
        print(error, file=sys.stderr)
    return 1 if errors else 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
TRAILER = "Co-Authored-By: Someone <someone@example.com>"


def run_hook(command: str, cwd: Path, tool: str = "Bash") -> subprocess.CompletedProcess:
    payload = {"tool_name": tool, "tool_input": {"command": command}, "cwd": str(cwd)}
    return subprocess.run([sys.executable, str(HOOK)], input=json.dumps(payload), capture_output=True, text=True)
```

`hooks/block_trailers.py`:
```python
#!/usr/bin/env python3
"""PreToolUse hook: deny git/svn commits whose message carries a trailer."""
from __future__ import annotations

import json
import re
import shlex
import sys
from pathlib import Path

TRAILER_RE = re.compile(
    r"(?i)(?:^|[\s\"'])(?:co-authored-by|signed-off-by|generated-by|claude-session|acked-by|"
    r"(?:review|reviewed|tested|reported|suggested|helped|patch|found)[ -]by):\s*\S"
)
GENERATED_RE = re.compile(r"(?i)generated with|🤖")
SHORT_SIGNOFF_RE = re.compile(r"-[A-Za-z]*s[A-Za-z]*")


def _tokens(command: str) -> list[str]:
    try:
        return shlex.split(command)
    except ValueError:
        return command.split()


def commit_args(command: str) -> tuple[str | None, list[str]]:
    """Return the VCS and the arguments after its commit subcommand, or (None, [])."""
    tokens = _tokens(command)
    for index, token in enumerate(tokens):
        if token == "commit" and "git" in tokens[:index]:
            return "git", tokens[index + 1 :]
        if token in ("commit", "ci") and "svn" in tokens[:index]:
            return "svn", tokens[index + 1 :]
    return None, []


def _message_files(args: list[str]) -> list[str]:
    files = []
    for index, arg in enumerate(args):
        if arg in ("-F", "--file") and index + 1 < len(args):
            files.append(args[index + 1])
        elif arg.startswith("--file="):
            files.append(arg.split("=", 1)[1])
    return files


def violations(command: str, cwd: str) -> list[str]:
    vcs, args = commit_args(command)
    if vcs is None:
        return []
    texts = [command]
    for name in _message_files(args):
        path = Path(cwd) / name
        if name != "-" and path.is_file():
            texts.append(path.read_text(encoding="utf-8", errors="replace"))
    found = []
    for text in texts:
        if TRAILER_RE.search(text):
            found.append("a trailer line (Co-Authored-By, Signed-off-by, ...)")
        if GENERATED_RE.search(text):
            found.append("a 'Generated with' attribution")
    if vcs == "git":
        for arg in args:
            if arg in ("--signoff", "--trailer") or arg.startswith("--trailer="):
                found.append(f"the {arg.split('=')[0]} flag")
            elif SHORT_SIGNOFF_RE.fullmatch(arg):
                found.append("the -s (sign-off) flag")
    return sorted(set(found))


def main() -> int:
    try:
        payload = json.load(sys.stdin)
    except json.JSONDecodeError:
        return 0
    if payload.get("tool_name") != "Bash":
        return 0
    command = payload.get("tool_input", {}).get("command", "")
    found = violations(command, payload.get("cwd", "."))
    if not found:
        return 0
    print(
        "Commit blocked: the message contains " + "; ".join(found) + ". "
        "A commit message is a subject line plus an optional body, with no trailers. "
        "Rewrite the message without them and commit again.",
        file=sys.stderr,
    )
    return 2


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""Copy cv-kit's rule files into a project's .claude/rules directory."""
from __future__ import annotations

import argparse
import shutil
import sys
from pathlib import Path

RULES_DIR = Path(__file__).resolve().parents[1] / "rules"


def install(rules_dir: Path, project: Path, force: bool = False) -> tuple[list[str], list[str]]:
    target = project / ".claude" / "rules"
    target.mkdir(parents=True, exist_ok=True)
    copied, skipped = [], []
    for source in sorted(rules_dir.glob("*.md")):
        destination = target / source.name
        if destination.exists() and not force:
            skipped.append(source.name)
            continue
        shutil.copyfile(source, destination)
        copied.append(source.name)
    return copied, skipped


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("project", type=Path)
    parser.add_argument("--force", action="store_true", help="overwrite files that already exist")
    args = parser.parse_args(argv)
    if not args.project.is_dir():
        print(f"{args.project}: not a directory", file=sys.stderr)
        return 1
    copied, skipped = install(RULES_DIR, args.project, args.force)
    print(f"copied: {', '.join(copied) or 'none'}")
    print(f"kept existing: {', '.join(skipped) or 'none'}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

`scripts/lint_research.py`:
```python
"""Reject a research claims ledger that has missing fields or numbers absent from their quote."""
from __future__ import annotations

import json
import re
import sys
from pathlib import Path

FIELDS = ("id", "value", "metric", "dataset", "split", "resolution", "hardware", "arxiv", "locator", "quote", "label")
ARXIV_RE = re.compile(r"^\d{4}\.\d{4,5}v\d+$")
LABELS = {"reported", "reproduced", "unverified"}


def lint_claims(claims: list[dict]) -> list[str]:
    errors: list[str] = []
    seen: set[str] = set()
    for index, claim in enumerate(claims):
        tag = str(claim.get("id") or f"#{index}")
        for field in FIELDS:
            if claim.get(field) in (None, ""):
                errors.append(f"{tag}: missing {field}")
        if tag in seen:
            errors.append(f"{tag}: duplicate id")
        seen.add(tag)
        if claim.get("arxiv") and not ARXIV_RE.match(str(claim["arxiv"])):
            errors.append(f"{tag}: arxiv {claim['arxiv']!r} must look like 2511.09554v2 (id plus version)")
        if claim.get("label") and claim["label"] not in LABELS:
            errors.append(f"{tag}: label {claim['label']!r} must be one of {sorted(LABELS)}")
        value = claim.get("value")
        if value not in (None, ""):
            if not isinstance(value, str):
                errors.append(f"{tag}: value must be a string copied from the quote")
            elif claim.get("quote") and value.strip() not in str(claim["quote"]):
                errors.append(f"{tag}: value {value!r} does not appear in its quote")
    return errors


def main(argv: list[str] | None = None) -> int:
    args = sys.argv[1:] if argv is None else argv
    if len(args) != 1:
        print("usage: lint_research.py claims.json", file=sys.stderr)
        return 2
    claims = json.loads(Path(args[0]).read_text(encoding="utf-8"))
    errors = lint_claims(claims)
    for error in errors:
        print(error, file=sys.stderr)
    return 1 if errors else 0


if __name__ == "__main__":
    raise SystemExit(main())
```
